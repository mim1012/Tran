## 📊 데이터 요구사항

### 자주 나가는 품목 쿼리
```sql
-- 최근 3개월 판매 빈도 순
SELECT
    p.Id,
    p.Code,
    p.Name,
    COUNT(si.Id) AS SaleCount,
    COALESCE(cp.Price, p.DefaultPrice) AS Price,
    COALESCE(i.ConfirmedQuantity, 0) AS StockQuantity,
    p.SafetyStockLevel
FROM Products p
INNER JOIN SaleItems si ON p.Id = si.ProductId
INNER JOIN Sales s ON si.SaleId = s.Id
LEFT JOIN CompanyPrices cp ON (p.Id = cp.ProductId AND cp.CompanyId = @CurrentCompanyId)
LEFT JOIN Inventory i ON p.Id = i.ProductId
WHERE s.ToCompanyId = @CurrentCompanyId
  AND s.CreatedAt >= DATEADD(MONTH, -3, GETDATE())
  AND p.TransactionType IN ('Sell', 'Both')
GROUP BY p.Id, p.Code, p.Name, cp.Price, p.DefaultPrice, i.ConfirmedQuantity, p.SafetyStockLevel
ORDER BY SaleCount DESC
LIMIT 10
```

### 판매 확정 시 처리 (★ 핵심)
```csharp
// 1. 재고 충분 여부 최종 확인
foreach (var item in saleItems)
{
    var inventory = await _dbContext.Inventory
        .FirstOrDefaultAsync(i => i.ProductId == item.ProductId);

    if (inventory == null || inventory.ConfirmedQuantity < item.Quantity)
    {
        throw new InsufficientStockException(
            $"{item.ProductName}: 재고 부족 (요청: {item.Quantity}, 재고: {inventory?.ConfirmedQuantity ?? 0})");
    }
}

// 2. Sale 생성
var sale = new Sale
{
    SaleNumber = GenerateSaleNumber(),
    FromCompanyId = currentCompanyId,
    ToCompanyId = customerCompanyId,
    State = SaleState.Confirmed,
    Items = saleItems,
    TotalAmount = totalAmount,
    CreatedAt = DateTime.Now
};
await _dbContext.Sales.AddAsync(sale);

// 3. 재고 즉시 차감
foreach (var item in saleItems)
{
    var inventory = await _dbContext.Inventory
        .FirstOrDefaultAsync(i => i.ProductId == item.ProductId);

    inventory.ConfirmedQuantity -= item.Quantity;
}

// 4. Document 자동 생성 (★ 통합 포인트)
var document = new Document
{
    DocumentNumber = GenerateDocumentNumber(),
    SaleId = sale.Id,  // ★ 연결
    DocumentType = "Sale",
    FromCompanyId = currentCompanyId,
    ToCompanyId = customerCompanyId,
    State = DocumentState.Draft,
    Items = saleItems.Select(si => new DocumentItem
    {
        ProductId = si.ProductId,
        ProductName = si.ProductName,
        Quantity = si.Quantity,
        UnitPrice = si.UnitPrice,
        Amount = si.Amount
    }).ToList(),
    CreatedAt = DateTime.Now
};

// ContentHash 계산
document.ContentHash = _hashService.CalculateHash(document);
await _dbContext.Documents.AddAsync(document);

// 5. 채권 생성
var receivable = new Receivable
{
    SaleId = sale.Id,
    CompanyId = customerCompanyId,
    Amount = totalAmount,
    DueDate = DateTime.Now.AddDays(30)
};
await _dbContext.Receivables.AddAsync(receivable);

await _dbContext.SaveChangesAsync();

// 6. 알림 반환
return new SaleConfirmResult
{
    Success = true,
    SaleId = sale.Id,
    SaleNumber = sale.SaleNumber,
    DocumentId = document.Id,
    DocumentNumber = document.DocumentNumber,
    TotalAmount = totalAmount
};
```

```csharp
public class StockValidator
{
    public StockValidationResult ValidateStock(int productId, int requestedQuantity)
    {
        var inventory = _dbContext.Inventory
            .FirstOrDefault(i => i.ProductId == productId);

        if (inventory == null || inventory.ConfirmedQuantity < requestedQuantity)
        {
            return new StockValidationResult
            {
                IsValid = false,
                Available = inventory?.ConfirmedQuantity ?? 0,
                Requested = requestedQuantity,
                Message = $"재고 부족 (요청: {requestedQuantity}, 재고: {inventory?.ConfirmedQuantity ?? 0})"
            };
        }

        if (inventory.ConfirmedQuantity - requestedQuantity < inventory.SafetyStockLevel)
        {
            return new StockValidationResult
            {
                IsValid = true,
                IsWarning = true,
                Message = "안전 재고 미만 경고"
            };
        }

        return new StockValidationResult { IsValid = true };
    }
}
```
